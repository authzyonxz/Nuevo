}

function aad(direction, path, context) {
  return Buffer.from([
    PROTOCOL,
    direction,
    'POST',
    path,
    String(context.version),
    context.keyId,
    context.clientNonceB64,
    String(context.timestamp),
    context.requestId,
    context.sessionId || ''
  ].join('|'), 'utf8');
}

function parsePayload(payload) {
  if (!isObject(payload)) throw protocolError(400, 'E_BAD_ENVELOPE', 'Payload criptográfico ausente.');
  const nonce = fromBase64Url(payload.nonce, 'payload.nonce');
  const ciphertext = fromBase64Url(payload.ciphertext, 'payload.ciphertext');
  const tag = fromBase64Url(payload.tag, 'payload.tag');
  if (nonce.length !== 12 || tag.length !== 16 || ciphertext.length > 1024 * 1024) {
    throw protocolError(400, 'E_BAD_ENVELOPE', 'Tamanho criptográfico inválido.');
  }
  return { nonce, ciphertext, tag };
}
